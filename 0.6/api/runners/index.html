    Runners¶

    BaseRunner ¶

    BaseRunner(run_options: RunOptions)

    Bases: ABC

    Abstract base class for all metapipeline runner backends.

    Subclasses register themselves by defining a RUNNER_NAME class attribute. The value is the string name used to select the runner::

    class MyRunner(BaseRunner):
    RUNNER_NAME = "my-runner"

    def run(self, g): ...      # returns RunResults
    def resume(self, g): ...

    Registration happens automatically at class-definition time via __init_subclass__. run_metapipeline looks up the concrete class from _REGISTRY, so a new runner becomes available as soon as its module is imported, with no changes to run_metapipeline needed.

    Subclasses should not override __init__. The RunOptions passed in is stored as self.run_options, available to every runner without repeating the constructor.

    RunOptions(
    runner_name: str = "python",
    out_dir: Path | None = None,

    nf_profile: str | None = None,
    nf_config: Path | None = None,

    resume: bool = False,
    input_mode: InputMode = FAIL,
    inputs: dict[str, Any] = dict(),
    prompter: Prompter | None = None,
)

    Runtime options that control how a metapipeline run is executed.

    Constructed from CLI arguments and passed to the runner. All fields have sensible defaults so the dataclass can be instantiated without arguments for programmatic use.

    Attributes:

    Name Type Description
    runner_name str

    Name of the runner backend to use. Must match a key registered in base_runner._REGISTRY.
    out_dir Path | None

    Directory every step runs in: step outputs with relative outdirs land here, plus Nextflow's work dir (work/) and nf-meta's state (.nf-meta/). None means nf-meta-out in the current directory. A relative path is relative to the current directory. Created if it does not exist.
    verbose bool

    Enables verbose logging output.
    output_lines int

    Number of most-recent output lines shown in the live scrolling panel while a step is running. Only has an effect in the Python runner.
    start str | None

    Step ID at which execution begins. Steps upstream of it are skipped. References to them are taken from their last runs, or treated as missing values (see input_mode). None means start from all source steps.
    target str | None

    Step ID at which execution ends. Steps downstream of it are skipped. None means run until all sink steps.
    nf_profile str | None

    Nextflow profile name (or comma-separated list of names) appended to every step's profiles, taking precedence over them. Whitespace is stripped before passing to Nextflow.
    nf_config Path | None

    Nextflow config file passed to every step (-c), after the global and the step's own configs, so it takes precedence.
    stub bool

    When True, passes -stub to every Nextflow invocation so that only the stub block of each process is executed. Useful for validating pipeline structure without real computation.
    resume bool

    When True, skips every step that already succeeded with the same step key (its definition, config and input files, globals and upstream steps), and starts the others with Nextflow's -resume against the step's own Nextflow cache.
    input_mode InputMode

    What happens with values the run needs but doesn't have (inputs without a value, references to steps outside the run): prompt, fail (default) or reproduce. See InputMode.
    inputs dict[str, Any]

    Values given for this run, by input name, or by <step_id>.<param> for a reference to a step outside the run.
    prompter Prompter | None

    Asks for a missing value in prompt mode (default: a plain input() prompt).

    def build_step_spec(
    wf: Workflow,
    global_options: GlobalOptions | None,
    *,
    state_dir: Path,
    launch_dir: Path,
    work_dir: Path | None = None,
    log_file: Path | None = None,
    nextflow_cache_dir: Path | None = None,
    project_dir: Path | None = None,
    schema: Schema | None = None,
    extra_profile: str = "",
    nf_config: Path | None = None,
    resume: bool = False,
    stub: bool = False,
) -> StepSpec:
    """Combine a step whose references are already resolved with the run's
    globals and locations into a `StepSpec`.

    With a `project_dir`, relative input paths in the step's `params_file` and
    in `globals.params` are made absolute against it (`paths.is_input_path`,
    using `schema`). The step's own params are taken as they are.

    Raises:
        NfMetaRunnerError: if the global config file doesn't exist.
    """
    params_file = getattr(wf, "params_file", None)
    params: dict[str, Any] = dict(load_params_file(params_file)) if params_file else {}
    if global_options and global_options.params:
        params.update(global_options.params)
    if project_dir is not None:
        params = absolutize_input_paths(params, project_dir, schema)
    params.update(wf.params or {})

    profile = getattr(wf, "nf_profile", None)
    profiles: list[str] = []
    if global_options and global_options.nf_profile and not profile:
        profiles.append(global_options.nf_profile)
    if profile:
        profiles.append(profile)
    if extra_profile:
        profiles.append(extra_profile)

    config_files: list[Path] = []
    if global_options and global_options.nf_config:
        global_cfg = Path(global_options.nf_config)
        if not global_cfg.exists():
            raise NfMetaRunnerError(f"Global config file does not exist: {global_cfg}")
        config_files.append(global_cfg)
    if wf.nf_config:
        config_files.append(Path(wf.nf_config))
    if nf_config is not None:
        config_files.append(Path(nf_config))

    return StepSpec(
        workflow=wf,
        params=params,
        profiles=tuple(profiles),
        config_files=tuple(config_files),
        launch_dir=launch_dir,
        work_dir=work_dir,
        state_dir=state_dir,
        log_file=log_file,
        nextflow_cache_dir=nextflow_cache_dir,
        resume=resume,
        stub=stub,
    )

    def engine_for(workflow_type: WorkflowType) -> Engine:
    """Return the engine registered for `workflow_type`.

    Raises:
        NfMetaRunnerError: if no engine is registered for it.
    """
    engine = _REGISTRY.get(WorkflowType(workflow_type))
    if engine is None:
        raise NfMetaRunnerError(
            f"No engine registered for step type '{workflow_type}'. Known types: {[str(t) for t in _REGISTRY]}"
        )
    return engine

    Built-in: Python runner¶

    SimplePythonRunner ¶

    SimplePythonRunner(run_options)

    Bases: BaseRunner

    The default runner (RUNNER_NAME = "python"). Executes each workflow as a nextflow run/nextflow module run subprocess directly from this Python process, in DAG order, streaming output as it runs.

    Every step runs Nextflow from the out dir (run_options.out_dir, default nf-meta-out in the current directory), sharing its work/ dir. Relative input paths are made absolute against the project dir first. nf-meta's own state lives in <out dir>/.nf-meta/.

    resume=True works on two levels: a step that already succeeded with an identical step key (its definition, the content of its config and params files, size and modification time of its local input files, the globals and all upstream step keys) is skipped; a step that runs is started with -resume, and each step has its own Nextflow cache dir, so Nextflow resumes that step's last run.

    def plan(self, graph: MetaworkflowGraph, resume: bool = False, dry_run: bool = False) -> RunPlan:
    """Resolve every selected step to its spec and command, in DAG order,
    without writing files or running anything. Values the run needs are
    collected first (see `RunOptions.input_mode`); in prompt mode this
    asks for them, even in a dry run.

    Raises:
        NfMetaRunnerError: if values are missing (except in a dry run
            outside prompt mode, where they become placeholders) or
            invalid, if no out dir can be determined, if a config loaded
            from a URL has relative input paths, or if two steps write to
            the same outdir.
    """
    self.out_dir = self._resolve_out_dir(graph)
    project_dir = graph.project_dir
    # params_file/nf_config are read from here on: as absolute paths
    global_options = graph.global_options and with_absolute_config_paths(graph.global_options, project_dir)
    workflows = [with_absolute_config_paths(wf, project_dir) for wf in self._selected_workflows(graph)]
    subset_ids = {wf.id for wf in workflows}

    # Resolved params and step keys per workflow id, seeded from cached runs
    # of excluded workflows
    resolved: dict[str, dict] = {}
    keys: dict[str, str] = {}
    for wf in graph.get_workflows():
        if wf.id not in subset_ids:
            wf = with_absolute_config_paths(wf, project_dir)
            keys[wf.id] = wf.hash()
            cached = self._load_resolved_params(wf)
            if cached is not None:
                resolved[wf.id] = cached

    requests = self._input_requests(graph, workflows, subset_ids, resolved)
    collected = collect(
        requests,
        self.run_options.inputs,
        InputMode(self.run_options.input_mode),
        project_dir=project_dir,
        prompter=self.run_options.prompter,
        dry_run=dry_run,
    )
    input_values = {r.name: collected.values[r.name] for r in requests if not r.is_reference}
    reference_values = {Binding.parse(r.name): collected.values[r.name] for r in requests if r.is_reference}

    planned: list[PlannedStep] = []
    for wf in workflows:
        bound = graph.bound_params(wf.id)
        if bound or input_refs(wf.params):
            wf = wf.model_copy(update={"params": substitute_inputs(wf.params, bound, input_values)})
        overrides = {b.param: v for b, v in reference_values.items() if b.step_id == wf.id}
        if overrides:
            wf = wf.model_copy(update={"params": {**(wf.params or {}), **overrides}})
        schema = param_schema(wf)
        if project_dir is not None and wf.params:
            wf = wf.model_copy(update={"params": absolutize_input_paths(wf.params, project_dir, schema)})
        elif project_dir is None:
            self._reject_relative_inputs(wf, schema)

        # Before references are resolved: other steps' outputs are covered by
        # their step keys, and may not exist yet
        global_params = (global_options.params if global_options else None) or {}
        if project_dir is not None and global_params:
            global_params = absolutize_input_paths(global_params, project_dir, schema)
        input_files = input_fingerprints({**global_params, **(wf.params or {})})
        wf_resolved = resolve_references(wf, graph, resolved=resolved)
        resolved[wf.id] = wf_resolved.params or {}

        upstream = [keys[p.id] for p in graph.predecessors(wf) if p.id in keys]
        key = self._step_key(wf_resolved, global_options, upstream, input_files)
        keys[wf.id] = key
        state_dir = self._state_dir(wf, key)

        spec = build_step_spec(
            wf_resolved,
            global_options,
            state_dir=state_dir,
            launch_dir=self.out_dir,
            work_dir=self.out_dir / "work",
            log_file=state_dir / "nextflow.log",
            nextflow_cache_dir=self._nextflow_cache_dir(wf),
            project_dir=project_dir,
            schema=schema,
            extra_profile=self.run_options.nf_profile or "",
            nf_config=self.run_options.nf_config,
            resume=resume,
            stub=self.run_options.stub,
        )
        planned.append(
            PlannedStep(
                # as configured, with the config files' absolute paths (as its resolved params' cache key)
                workflow=with_absolute_config_paths(graph.get_workflow_by_id(wf.id) or wf, project_dir),
                resolved_params=wf_resolved.params or {},
                spec=spec,
                command=engine_for(spec.workflow_type).build(spec),
                # The step's outdir holds the last run's results, so an older
                # success with this key counts only if no other key ran since.
                skip=resume and self._check_run_success(run_dir=state_dir) and self._last_run_was(wf, state_dir),
            )
        )
    _check_outdirs(planned)
    return RunPlan(
        steps=planned,
        values=dict(collected.values),
        missing=[r.name for r in collected.missing],
    )
