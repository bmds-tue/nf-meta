    Python API¶

    Build, edit, validate and run metapipelines from Python. Everything here is importable from nf_meta directly (from nf_meta import Metapipeline, NfPipeline, Ref, run_step).

    A guide with worked examples will follow; until then, the docstrings below describe each call.

    Metapipeline¶

    Metapipeline ¶

    Metapipeline(project_dir: str | Path | None = None)

    A metapipeline being built or edited, which can be saved, planned and run.

    Every change is validated when it is made; an invalid change raises (ConfigValidationError or GraphValidationError) and leaves the metapipeline unchanged. Nothing is written to disk before save().

    Steps are NfPipeline/NfModule models. The metapipeline keeps its own copy of each: to change a step, change the model (the one passed to add, or one from steps or mp[step_id]) and pass it to update.

    mp = Metapipeline()
fetch = mp.add(NfPipeline(name="nf-core/fetchngs", version="1.12.0",
                          params={"input": "ids.csv", "outdir": "fetchngs"}))
mp.add(NfPipeline(name="nf-core/rnaseq", version="3.26.0", params={
    "input": Ref(fetch, "outdir") / "samplesheet/samplesheet.csv",
    "outdir": "rnaseq"}))
mp.save("metaconfig.yaml")
results = mp.run(out_dir="results")

    @classmethod
def load(cls, config: str | Path) -> "Metapipeline":
    """Load and validate the config at `config` (a path or an `http(s)://`
    URL). Its folder becomes the project dir. Warnings found while loading
    are issued with `warnings.warn`."""
    mp = cls.__new__(cls)
    mp._session = EditorSession(write_on_change=False)
    mp._session.load_config(config)
    for d in mp.graph.warnings:
        warnings.warn(f"{config}: {d.yaml_path}: {d.message}", stacklevel=2)
    return mp

    References and inputs¶

    Ref ¶

    Bases: str

    A reference to another step's param or to an input, usable as a param value. It is a str holding the reference syntax, and / appends a path:

    Ref(fetch, "outdir") / "samplesheet/samplesheet.csv"
# '${n01:params:outdir}/samplesheet/samplesheet.csv'
Ref.input("ids")
# '${inputs:ids}' (a param's whole value: no `/`)

    step is a step model or a step ID.

    def keywords(self) -> dict[str, Any]:

    values = {k: getattr(self, k) for k in SCHEMA_KEYWORDS if k != "schema"}

    def model_dump_config(self) -> dict:
    """This input as written in a config file."""
    data = self.model_dump(exclude_none=True, exclude_defaults=True, by_alias=True)
    if "schema_" in data:
        data["schema"] = data.pop("schema_")
    if "bind" in data:
        data["bind"] = list(data["bind"])
    if "enum" in data:
        data["enum"] = list(data["enum"])
    return data

    @classmethod
def from_config(cls, data: Any) -> "InputDefinition":
    """Parse an `inputs:` entry. `None` stands for an input with no keys."""
    data = dict(data or {})
    if "schema" in data:
        data["schema_"] = data.pop("schema")
    return cls(**data)

    def run_step(step: Workflow, *, project_dir: Path | None = None, **options: Any) -> RunResults:
    """Run a single step (an `NfPipeline` or `NfModule`) as a one-step
    metapipeline and return the results. `options` are `RunOptions` fields
    (`out_dir=`, `nf_profile=`, `resume=`, ...). Relative input paths resolve
    against `project_dir` (default: the current directory).

    Raises `ConfigValidationError` if the step is invalid, and
    `NfMetaRunnerError` if it fails.
    """
    g = MetaworkflowGraph()
    g.project_dir = (project_dir or Path.cwd()).resolve()
    g.add_workflow(step)
    return run_metapipeline(g, RunOptions(**options))

    @classmethod
def from_planned(cls, step: PlannedStep) -> "StepResult":
    """The result of running `step` as planned."""
    outdir = step.spec.params.get("outdir")
    if isinstance(outdir, str) and outdir and "://" not in outdir:
        outdir = step.spec.launch_dir / outdir
    return cls(
        step_id=step.workflow.id,
        name=step.workflow.name,
        params=step.spec.params,
        outdir=Path(outdir) if isinstance(outdir, (str, Path)) and outdir else None,
        state_dir=step.spec.state_dir,
        log_file=step.spec.log_file,
        command=step.command.argv,
        skipped=step.skip,
    )
